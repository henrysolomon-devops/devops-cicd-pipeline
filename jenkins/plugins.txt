# The minimal set of plugins this pipeline actually needs, nothing more.
# One plugin per line, comments starting with # are ignored by the
# plugin installation manager tool that reads this file.

# JCasC itself - without this, jenkins.yaml would just sit there unread.
configuration-as-code

# Lets Jenkins check out this repo over plain HTTPS (it's public, so no
# credential is needed just to read it).
git

# Pulls in Pipeline: Groovy, Pipeline: Basic Steps, and Pipeline: Stage
# View together, everything the Jenkinsfile itself needs to run as a
# multi-stage pipeline with input approval gates.
workflow-aggregator

# Lets JCasC register the pipeline job itself (pointing at this repo
# and its Jenkinsfile) as code, instead of clicking "New Item" by hand.
job-dsl

# Gives the Jenkinsfile a withCredentials step, used to pull the GHCR
# token out of the credential JCasC defines, rather than the pipeline
# reading the secret file directly.
credentials-binding

# Nicer, more idiomatic pipeline syntax for the Docker build/push stage
# (docker.build(), docker.withRegistry()) instead of raw shell calls.
docker-workflow
